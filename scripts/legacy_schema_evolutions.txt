# Legacy-monolith schema objects that the service migration chains INTENTIONALLY changed (each line: the start of the legacy
# inventory line, tab-separated: table, kind, object). verify-institute-services-on-postgres.sh fails if any OTHER legacy object
# is missing from the service-chain schema; additions are allowed (later revisions are additive).
# enrollment 0002_enrollment_student_identity: enrollments reference the global Student ID without a foreign key
enrollments	constraint	enrollments_student_id_fkey
# paper 0002_paper_question_bank: platform (GLOBAL) questions and papers have no provider/branch
bank_questions	column	branch_id uuid NO
bank_questions	column	provider_id uuid NO
papers	column	branch_id uuid NO
papers	column	provider_id uuid NO
# (PostgreSQL 18 also reports the same NOT NULL rules as named constraints)
bank_questions	constraint	bank_questions_branch_id_not_null
bank_questions	constraint	bank_questions_provider_id_not_null
papers	constraint	papers_branch_id_not_null
papers	constraint	papers_provider_id_not_null
# exam 0004_exam_tests_audience: DRAFT plans, platform papers in exams, exam settings in the append-only trigger
exam_plans	constraint	exam_plans_status_check
exam_version_papers	column	source_provider_id uuid NO
exam_version_papers	constraint	exam_version_papers_source_provider_id_not_null
exam_versions	trigger	exam_versions_immutable
# paper 0010_optional_question_marks: marks belong to the test; a question may have no marks of its own (both NULL), so a paper
# may have no total
paper_versions	column	total_marks numeric NO
question_versions	column	marks_negative numeric NO
question_versions	column	marks_positive numeric NO
question_versions	constraint	question_versions_marks_check
paper_versions	constraint	paper_versions_total_marks_not_null
question_versions	constraint	question_versions_marks_negative_not_null
question_versions	constraint	question_versions_marks_positive_not_null
# paper 0015_question_types: MSQ and NAT have no correct letter (their key is answer_key), so correct_option becomes nullable and
# the letter CHECK becomes part of question_versions_shape_check (an MCQ still needs one of A-D)
question_versions	column	correct_option character varying NO
question_versions	constraint	question_versions_correct_check
